# SPIKE CODE (S6) - throwaway, see docs/spikes/S6-network-libraries.md. Do not build on this.

# Network library spike: ENet + libjuice + libsodium vs GameNetworkingSockets, fetched at pinned
# upstream tags (exact commits are recorded in the report) and built inside the shadPS4 tree with
# the fork's toolchain. Only the three probe executables are built (cmake --build --target ...).
include(FetchContent)

# The top-level CMakeLists.txt calls include_directories(src); the probes and the fetched libraries
# do not need the emulator headers, so drop the inherited list for everything in this directory.
set_property(DIRECTORY PROPERTY INCLUDE_DIRECTORIES "")

FetchContent_Declare(enet
    GIT_REPOSITORY https://github.com/lsalzman/enet.git
    GIT_TAG v1.3.18             # 2662c0de09e36f2a2030ccc2c528a3e4c9e8138a
    GIT_SHALLOW TRUE)
FetchContent_Declare(libjuice
    GIT_REPOSITORY https://github.com/paullouisageneau/libjuice.git
    GIT_TAG v1.7.4              # b89c792e3612faf2f12cf35bcc56857313a06be3
    GIT_SHALLOW TRUE)
# libsodium-cmake has no releases and no tags: pinned to the master head of 2026-05-29.
# It wraps jedisct1/libsodium as a submodule at 1.0.20-FINAL (93a7d0d41fe2e32409b5d00386946f491750b7de).
FetchContent_Declare(sodium
    GIT_REPOSITORY https://github.com/robinlinden/libsodium-cmake.git
    GIT_TAG 9b2848dfc1b917a9410f0de9d81059b26cbfaa8d)
# GNS: the client side of a direct-IP connection always binds its UDP socket to the wildcard
# address and there is no option for a local address. The spike patch binds it to 127.0.0.1 when
# the remote is loopback, so nothing in the probe listens beyond loopback (0C-R8).
# GIT_SUBMODULES "": GNS's submodules (webrtc ~700 MB, abseil 2023, vjson) are only needed for
# USE_STEAMWEBRTC / the cert tool; FetchContent would otherwise clone them all recursively.
FetchContent_Declare(gns
    GIT_REPOSITORY https://github.com/ValveSoftware/GameNetworkingSockets.git
    GIT_TAG v1.6.0              # 2cb93a06350bb065db53abdb0d87cf297e0bfd34
    GIT_SHALLOW TRUE
    GIT_SUBMODULES ""
    UPDATE_DISCONNECTED TRUE
    PATCH_COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_LIST_DIR}/gns_loopback_bind.cmake)

option(BBCOOP_SPIKE_S6_PROTOBUF_REDIRECT "S6: point GNS's find_package(Protobuf) at the in-tree protobuf" ON)

set(NO_TESTS ON CACHE BOOL "" FORCE)
set(SODIUM_DISABLE_TESTS ON CACHE BOOL "" FORCE)
string(TIMESTAMP _s6_t0 "%s")
FetchContent_MakeAvailable(enet libjuice sodium)
string(TIMESTAMP _s6_t1 "%s")
# libsodium-cmake bundles libsodium 1.0.20, whose AVX-512 Argon2 kernel uses
# '#pragma clang attribute push(target("...,avx512f,evex512"))'. clang 22 dropped the evex512
# feature, ignores the whole attribute and then rejects the AVX-512 intrinsics
# ("always_inline function '_mm512_loadu_si512' requires target feature 'avx512f'").
# libsodium 1.0.21+ guards it with __clang_major__ < 22; libsodium-cmake has not moved yet.
# Compile that one file with -mavx512f, as libsodium's own autotools build does (runtime dispatch).
set_source_files_properties(
    ${sodium_SOURCE_DIR}/libsodium/src/libsodium/crypto_pwhash/argon2/argon2-fill-block-avx512f.c
    TARGET_DIRECTORY sodium PROPERTIES COMPILE_OPTIONS "-mavx512f")
math(EXPR _s6_light_s "${_s6_t1} - ${_s6_t0}")
message(STATUS "S6: enet + libjuice + libsodium fetch+configure took ${_s6_light_s} s")

add_executable(bbcoop_s6_enet_echo enet_echo.cpp udp_rebind_proxy.h spike_common.h)
# ENet only sets include_directories() in its own directory; consumers add the include root.
target_include_directories(bbcoop_s6_enet_echo PRIVATE ${enet_SOURCE_DIR}/include)
target_link_libraries(bbcoop_s6_enet_echo PRIVATE enet sodium ws2_32 winmm iphlpapi)

add_executable(bbcoop_s6_juice_ice juice_ice.cpp spike_common.h)
target_link_libraries(bbcoop_s6_juice_ice PRIVATE LibJuice::LibJuiceStatic ws2_32 iphlpapi)

# GameNetworkingSockets: Windows-native crypto, protobuf from the shadPS4 tree.
set(USE_CRYPTO BCrypt CACHE STRING "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)

# GNS calls find_package(Protobuf) (config mode first, then the FindProtobuf module). shadPS4 builds
# protobuf from externals/protobuf with add_subdirectory, so no package config exists. A redirect
# config (CMake >= 3.24 searches CMAKE_FIND_PACKAGE_REDIRECTS_DIR first) points GNS at the in-tree
# targets: protobuf::libprotobuf for linking and protobuf::protoc for code generation.
if (BBCOOP_SPIKE_S6_PROTOBUF_REDIRECT)
    file(WRITE ${CMAKE_FIND_PACKAGE_REDIRECTS_DIR}/protobuf-config.cmake
"# Generated by the S6 spike: use the protobuf targets of the shadPS4 tree.
if (NOT TARGET protobuf::libprotobuf OR NOT TARGET protobuf::protoc)
    set(Protobuf_FOUND FALSE)
    set(Protobuf_NOT_FOUND_MESSAGE \"shadPS4 in-tree protobuf targets are missing\")
    return()
endif()
include(\"${CMAKE_SOURCE_DIR}/externals/protobuf/cmake/protobuf-generate.cmake\")
# Same contract as FindProtobuf's protobuf_generate_cpp (PROTOBUF_GENERATE_CPP_APPEND_PATH defaults
# to TRUE there): each .proto's own directory is an import root, outputs go to the binary dir.
function(protobuf_generate_cpp SRCS HDRS)
    set(_out)
    protobuf_generate(APPEND_PATH LANGUAGE cpp OUT_VAR _out PROTOS \${ARGN})
    set(_srcs)
    set(_hdrs)
    foreach(_f \${_out})
        if (_f MATCHES \"cc$\")
            list(APPEND _srcs \${_f})
        else()
            list(APPEND _hdrs \${_f})
        endif()
    endforeach()
    set(\${SRCS} \${_srcs} PARENT_SCOPE)
    set(\${HDRS} \${_hdrs} PARENT_SCOPE)
endfunction()
")
    file(WRITE ${CMAKE_FIND_PACKAGE_REDIRECTS_DIR}/protobuf-config-version.cmake
        "set(PACKAGE_VERSION 0)\nset(PACKAGE_VERSION_COMPATIBLE TRUE)\n")
endif()

# GNS's install(EXPORT) would need libprotobuf in an export set (shadPS4 builds protobuf with
# protobuf_INSTALL OFF): "install(EXPORT ...) includes target GameNetworkingSockets_s which requires
# target libprotobuf that is not in any export set". Skip install rules in this directory scope.
set(CMAKE_SKIP_INSTALL_RULES ON)
# shadPS4 adds WIN32_LEAN_AND_MEAN to every directory (add_compile_definitions in the top-level
# CMakeLists.txt). GNS relies on <windows.h> pulling in timeapi.h (::timeBeginPeriod in
# steamnetworkingsockets_socketthread.cpp), so drop the define for the GNS subtree.
get_directory_property(_s6_defs_saved COMPILE_DEFINITIONS)
set(_s6_defs ${_s6_defs_saved})
list(REMOVE_ITEM _s6_defs WIN32_LEAN_AND_MEAN)
set_property(DIRECTORY PROPERTY COMPILE_DEFINITIONS "${_s6_defs}")
string(TIMESTAMP _s6_t2 "%s")
FetchContent_MakeAvailable(gns)
set_property(DIRECTORY PROPERTY COMPILE_DEFINITIONS "${_s6_defs_saved}")
string(TIMESTAMP _s6_t3 "%s")
math(EXPR _s6_gns_s "${_s6_t3} - ${_s6_t2}")
message(STATUS "S6: GameNetworkingSockets fetch+configure took ${_s6_gns_s} s")

add_executable(bbcoop_s6_gns_echo gns_echo.cpp udp_rebind_proxy.h spike_common.h)
target_link_libraries(bbcoop_s6_gns_echo PRIVATE GameNetworkingSockets::static ws2_32 iphlpapi)
